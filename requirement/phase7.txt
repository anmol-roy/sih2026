src/formulation/schemas.py
from enum import Enum
from typing import List
from pydantic import BaseModel


class FormulationType(str, Enum):

    AYURVEDA = "ayurveda"
    SIDDHA = "siddha"
    UNANI = "unani"
    YOGA = "yoga"
    TRADITIONAL_KNOWLEDGE = "traditional_knowledge"
    MODERN_PHARMACEUTICAL = "modern_pharmaceutical"
    BIOLOGICAL_RESOURCE = "biological_resource"
    MIXED = "mixed"
    UNKNOWN = "unknown"


class Ingredient(BaseModel):

    name: str

    scientific_name: str | None = None

    biological_resource: bool = False

    traditional_use_indicator: bool = False


class FormulationClassification(BaseModel):

    formulation_type: FormulationType

    ingredients: List[Ingredient]

    traditional_knowledge_indicators: List[str]

    biological_resource_indicators: List[str]

    confidence: float

10. Build abs_checker.py

Create:

src/abs/abs_checker.py

Logic:

def assess_abs(formulation):

    biological_resources = []

    for ingredient in formulation.ingredients:

        if ingredient.biological_resource:

            biological_resources.append(
                ingredient.name
            )

    tk_detected = (
        len(formulation.traditional_knowledge_indicators) > 0
        or any(
            ingredient.traditional_use_indicator
            for ingredient in formulation.ingredients
        )
    )

    potentially_relevant = (
        len(biological_resources) > 0
    )

    reasons = []

    if biological_resources:

        reasons.append(
            "Biological resources were identified."
        )

    if tk_detected:

        reasons.append(
            "Traditional-knowledge indicators were identified."
        )

    return {
        "potentially_relevant": potentially_relevant,

        "biological_resources":
            biological_resources,

        "traditional_knowledge_detected":
            tk_detected,

        "reasons":
            reasons,

        "requires_human_review":
            potentially_relevant
    }

This is deliberately conservative.

15. Build a single formulation analysis function

Create:

src/formulation/analyze.py

Pipeline:

def analyze_formulation(text):

    extracted = extract_formulation(text)

    classification = classify_formulation(
        extracted
    )

    abs_result = assess_abs(
        classification
    )

    tk_results = search_traditional_knowledge(
        classification
    )

    patent_results = search_prior_art(
        classification
    )

    legal_results = search_relevant_law(
        classification,
        abs_result
    )

    return {
        "classification": classification,
        "abs": abs_result,
        "traditional_knowledge": tk_results,
        "prior_art": patent_results,
        "legal": legal_results
    }

Now you have one reusable backend capability.
